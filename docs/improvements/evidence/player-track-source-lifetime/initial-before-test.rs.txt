fn root() -> PathBuf {
    PathBuf::from(env!("CARGO_MANIFEST_DIR"))
        .join("tests/fixtures/playback-errors/player-track-source-lifetime")
}
fn bytes(n: &str) -> Vec<u8> {
    std::fs::read(root().join(n)).unwrap()
}
fn tick(p: &mut Player, ctx: &egui::Context) {
    let mut out = ctx.run_ui(Default::default(), |ui| p.present(ui.ctx()));
    out.textures_delta.clear();
}
#[derive(Default)]
struct Trace {
    pcm: Vec<u8>,
    flushes: Vec<Duration>,
    pauses: usize,
    resumes: usize,
    drops: usize,
}
struct Device {
    clock: Arc<AtomicU64>,
    trace: Arc<Mutex<Trace>>,
    fail: Arc<AtomicBool>,
    running: bool,
    anchor: Duration,
    frontier: Duration,
}
impl crate::audio::AudioBackend for Device {
    fn start(&mut self, s: crate::audio::AudioSpec) -> Result<(), crate::audio::AudioError> {
        assert_eq!((s.sample_rate, s.channels), (8000, 1));
        Ok(())
    }
    fn position(&self) -> Duration {
        if self.running {
            Duration::from_nanos(self.clock.load(Ordering::SeqCst))
        } else {
            self.anchor
        }
    }
    fn push(&mut self, p: crate::audio::AudioPacket) -> Result<(), crate::audio::AudioError> {
        self.frontier =
            p.presentation_time() + Duration::from_nanos(p.data.len() as u64 / 4 * 125000);
        self.trace.lock().unwrap().pcm.extend(p.data);
        Ok(())
    }
    fn flush(&mut self, t: Duration) -> Result<(), crate::audio::AudioError> {
        self.trace.lock().unwrap().flushes.push(t);
        self.anchor = t;
        self.frontier = t; // Flush discards the previous queued output.
        self.clock.store(t.as_nanos() as u64, Ordering::SeqCst);
        Ok(())
    }
    fn pause(&mut self) -> Result<(), crate::audio::AudioError> {
        self.anchor = self.position();
        self.running = false;
        self.trace.lock().unwrap().pauses += 1;
        Ok(())
    }
    fn resume(&mut self) -> Result<(), crate::audio::AudioError> {
        self.running = true;
        self.trace.lock().unwrap().resumes += 1;
        Ok(())
    }
    fn stop(&mut self) -> Result<(), crate::audio::AudioError> {
        self.running = false;
        Ok(())
    }
    fn drain_status(&self) -> Result<bool, crate::audio::AudioError> {
        if self.fail.load(Ordering::SeqCst) {
            Err(crate::audio::AudioError::PlaybackError(
                "synthetic output drain failure".into(),
            ))
        } else {
            Ok(self.position() >= self.frontier)
        }
    }
}

fn until(mut ready:impl FnMut()->bool){let end=Instant::now()+Duration::from_secs(5);while !ready(){assert!(Instant::now()<end,"native track clock timed out");std::thread::sleep(Duration::from_millis(1));}}
fn factory(clock:Arc<AtomicU64>,trace:Arc<Mutex<Trace>>)->impl FnOnce()->Box<dyn crate::audio::AudioBackend>+Send+'static{move||Box::new(Device{clock,trace,fail:Arc::new(AtomicBool::new(false)),running:false,anchor:Duration::ZERO,frontier:Duration::ZERO})}

impl Drop for Device{fn drop(&mut self){self.trace.lock().unwrap().drops+=1;}}
static TEMP_SEQUENCE:AtomicU64=AtomicU64::new(0);
struct Temp(PathBuf);
impl Temp{
    fn new()->Self{let path=std::env::temp_dir().join(format!("fvid-track-source-lifetime-{}-{}",std::process::id(),TEMP_SEQUENCE.fetch_add(1,Ordering::SeqCst)));std::fs::create_dir(&path).unwrap();Self(path)}
    fn media(&self)->PathBuf{self.0.join("live.mp4")}
    fn moved(&self)->PathBuf{self.0.join("renamed.mp4")}
    fn seed(&self){std::fs::write(self.media(),bytes("audio-only.mp4")).unwrap();}
    fn move_away(&self){std::fs::rename(self.media(),self.moved()).unwrap();assert!(!self.media().exists());}
}
impl Drop for Temp{fn drop(&mut self){std::fs::remove_dir_all(&self.0).unwrap();}}
fn open(temp:&Temp,nth:usize,memory:bool)->(Player,Arc<AtomicU64>,Arc<Mutex<Trace>>){
    let clock=Arc::new(AtomicU64::new(75_000_000));let trace=Arc::new(Mutex::new(Trace::default()));let mut p=Player{queue:vec![temp.media()],opened:Some(temp.media()),duration:Some(Duration::from_millis(600)),seekable:true,paused:false,..Default::default()};
    if memory{
        let source=crate::playback_mp4_audio::Mp4AudioReader::open_at(Cursor::new(bytes("audio-only.mp4")),Default::default(),nth).unwrap();
        p.audio_tracks=crate::audio::AudioStream::audio_tracks(&source);p.audio_track=nth;p.sound_codec=sound_codec(crate::audio::AudioStream::codec(&source));
        p.audio=Some(crate::audio_thread::AudioPlayback::start(Box::new(source),factory(clock.clone(),trace.clone())));
    }else{assert!(p.try_start_audio_choice_with_backend(&temp.media(),Some(nth),factory(clock.clone(),trace.clone())));}
    p.audio.as_ref().unwrap().play();until(||p.audio.as_ref().unwrap().position()==Duration::from_millis(75));assert_eq!(p.audio_tracks.len(),2);(p,clock,trace)
}
fn finish(p:&mut Player,ctx:&egui::Context,clock:&AtomicU64){clock.store(600_000_000,Ordering::SeqCst);until(||{tick(p,ctx);p.ended});assert!(p.audio_ended);assert!(p.error.is_none());}
fn check_new_pcm(p:&mut Player,ctx:&egui::Context,clock:&AtomicU64,trace:&Mutex<Trace>,nth:usize){until(||p.audio.as_ref().unwrap().position()==Duration::from_millis(75));until(||!trace.lock().unwrap().pcm.is_empty());let actual=trace.lock().unwrap().pcm.clone();let oracle=bytes(&format!("track-{nth}.f32le"));assert_eq!(actual,&oracle[2400..2400+actual.len()]);finish(p,ctx,clock);assert_eq!(trace.lock().unwrap().pcm,&oracle[2400..]);}

fn moved(from:usize,forward:bool,paused:bool){
    let temp=Temp::new();temp.seed();let(mut p,_old_clock,old)=open(&temp,from,false);let ctx=egui::Context::default();if paused{p.toggle_pause();until(||old.lock().unwrap().pauses==1);}temp.move_away();
    let clock=Arc::new(AtomicU64::new(0));let trace=Arc::new(Mutex::new(Trace::default()));p.cycle_audio_track_with_backend(forward,factory(clock.clone(),trace.clone()));assert!(p.audio.is_some(),"track switch closed the last shared descriptor before opening the moved source");assert_eq!(p.audio_track,1-from);assert_eq!(p.paused,paused);let drops=old.lock().unwrap().drops;assert_eq!(drops,1);
    until(||p.audio.as_ref().unwrap().position()==Duration::from_millis(75));if paused{assert_eq!(trace.lock().unwrap().resumes,0);assert!(trace.lock().unwrap().pcm.is_empty());p.toggle_pause();}check_new_pcm(&mut p,&ctx,&clock,&trace,1-from);
}
