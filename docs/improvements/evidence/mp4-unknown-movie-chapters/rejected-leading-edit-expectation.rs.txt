use std::io::Cursor;
fn bytes(n: &str) -> Vec<u8> {
    std::fs::read(std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
        .join("tests/fixtures/playback-errors/mp4-unknown-movie-chapters").join(n)).unwrap()
}
fn cases(clock: &str) -> Vec<String> {
    serde_json::from_slice::<serde_json::Value>(&bytes("manifest.json")).unwrap()[clock]
        .as_array().unwrap().iter().map(|s|s.as_str().unwrap().to_owned()).collect()
}
fn starts(n: &str) -> Vec<u64> {
    let r = fvid::container::mp4::Mp4Reader::open(Cursor::new(bytes(n)), Default::default()).unwrap();
    r.chapters().iter().map(|c|c.start_ns).collect()
}
#[cfg(feature="player")]
#[test]
fn edited_chapter_seeks_match_the_known_duration_oracle() {
    use std::time::Duration;
    for version in [0, 1] {
        let mut known = fvid::playback_native::NativeReader::software(Cursor::new(bytes(&format!("edited-v{version}-known-movie_clock.mp4"))), 32 << 20).unwrap();
        let n = format!("edited-v{version}-unknown-movie_clock.mp4");
        let mut actual = fvid::playback_native::NativeReader::software(Cursor::new(bytes(&n)), 32 << 20).unwrap();
        assert_eq!(known.duration(), Some(Duration::from_millis(5060)));
        assert_eq!(actual.duration(), known.duration());
        for ns in starts(&n) {
            let at = Duration::from_nanos(ns);
            let expected = known.seek_raw(at).unwrap().unwrap().into_rgb(32 << 20).unwrap();
            assert_eq!(expected, vec![130;16*16*3]);
            assert_eq!(actual.seek_raw(at).unwrap().unwrap().into_rgb(32 << 20).unwrap(), expected);
            assert_eq!(actual.frame_interval(), known.frame_interval());
        }
    }
}
