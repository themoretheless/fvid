#[derive(Default)]
struct Trace {
    pcm: Vec<u8>,
    flushes: Vec<Duration>,
    pauses: usize,
    resumes: usize,
    levels: Vec<bool>,
}
struct Device {
    clock: Arc<AtomicU64>,
    trace: Arc<Mutex<Trace>>,
    fail: Arc<AtomicBool>,
    running: bool,
    anchor: Duration,
    frontier: Duration,
}
impl crate::audio::AudioBackend for Device {
    fn start(&mut self, s: crate::audio::AudioSpec) -> Result<(), crate::audio::AudioError> {
        assert_eq!((s.sample_rate, s.channels), (8000, 1));
        Ok(())
    }
    fn position(&self) -> Duration {
        if self.running {
            Duration::from_nanos(self.clock.load(Ordering::SeqCst))
        } else {
            self.anchor
        }
    }
    fn push(&mut self, p: crate::audio::AudioPacket) -> Result<(), crate::audio::AudioError> {
        self.frontier =
            p.presentation_time() + Duration::from_nanos(p.data.len() as u64 / 4 * 125000);
        self.trace.lock().unwrap().pcm.extend(p.data);
        Ok(())
    }
    fn flush(&mut self, t: Duration) -> Result<(), crate::audio::AudioError> {
        self.trace.lock().unwrap().flushes.push(t);
        self.anchor = t;
        self.frontier = t; // Flush discards the previous queued output.
        self.clock.store(t.as_nanos() as u64, Ordering::SeqCst);
        Ok(())
    }
    fn pause(&mut self) -> Result<(), crate::audio::AudioError> {
        self.anchor = self.position();
        self.running = false;
        self.trace.lock().unwrap().pauses += 1;
        Ok(())
    }
    fn resume(&mut self) -> Result<(), crate::audio::AudioError> {
        self.running = true;
        self.trace.lock().unwrap().resumes += 1;
        Ok(())
    }
    fn set_volume(&mut self,_milli:u32)->Result<(),crate::audio::AudioError>{self.trace.lock().unwrap().levels.push(self.running);Ok(())}
    fn stop(&mut self) -> Result<(), crate::audio::AudioError> {
        self.running = false;
        Ok(())
    }
    fn drain_status(&self) -> Result<bool, crate::audio::AudioError> {
        if self.fail.load(Ordering::SeqCst) {
            Err(crate::audio::AudioError::PlaybackError(
                "synthetic output drain failure".into(),
            ))
        } else {
            Ok(self.position() >= self.frontier)
        }
    }
}

use std::io::{self,BufRead,Read,Seek,SeekFrom};
use std::sync::Condvar;
#[derive(Default)]
struct State {
    armed: bool,
    blocked: bool,
    released: bool,
}
#[derive(Default)]
struct Gate {
    state: Mutex<State>,
    wake: Condvar,
}
impl Gate {
    fn arm(&self) {
        self.state.lock().unwrap().armed = true;
    }
    fn blocked(&self) -> bool {
        self.state.lock().unwrap().blocked
    }
    fn release(&self) {
        self.state.lock().unwrap().released = true;
        self.wake.notify_all();
    }
}
struct Release(Option<Arc<Gate>>);
impl Drop for Release {
    fn drop(&mut self) {
        if let Some(gate) = &self.0 { gate.release(); }
    }
}
struct Input {
    cursor: Cursor<Vec<u8>>,
    gate: Arc<Gate>,
    range: std::ops::Range<u64>,
}
impl Input {
    fn wait(&self) -> io::Result<()> {
        if !self.range.contains(&self.cursor.position()) {
            return Ok(());
        }
        let mut s = self.gate.state.lock().unwrap();
        while s.armed && !s.released {
            s.blocked = true;
            self.gate.wake.notify_all();
            let (next, timeout) = self
                .gate
                .wake
                .wait_timeout(s, Duration::from_secs(10))
                .unwrap();
            s = next;
            if timeout.timed_out() && !s.released {
                return Err(io::Error::new(
                    io::ErrorKind::TimedOut,
                    "test input gate was not released",
                ));
            }
        }
        Ok(())
    }
}
impl Read for Input {
    fn read(&mut self, out: &mut [u8]) -> io::Result<usize> {
        if out.is_empty() {
            return Ok(0);
        }
        self.wait()?;
        let n = out.len().min(1);
        self.cursor.read(&mut out[..n])
    }
}
impl BufRead for Input {
    fn fill_buf(&mut self) -> io::Result<&[u8]> {
        self.wait()?;
        let n = (self.cursor.position() as usize).min(self.cursor.get_ref().len());
        let end = (n + 1).min(self.cursor.get_ref().len());
        Ok(&self.cursor.get_ref()[n..end])
    }
    fn consume(&mut self, n: usize) {
        self.cursor.consume(n);
    }
}
impl Seek for Input {
    fn seek(&mut self, from: SeekFrom) -> io::Result<u64> {
        self.cursor.seek(from)
    }
}
