def opus(channels=1,lengths=(6,),serial=900):
    head=b'OpusHead'+struct.pack('<BBHIhB',1,channels,0,48000,0,0);tags=b'OpusTags'+struct.pack('<I',4)+b'fvid'+struct.pack('<I',0)
    data=ogg_page(head,0,2,0,serial)+ogg_page(tags,0,0,1,serial)
    for i,size in enumerate(lengths):data+=ogg_page(bytes.fromhex('f8fffe')+bytes(size-3),(i+1)*960,4 if i+1==len(lengths)else 0,i+2,serial)
    return data
